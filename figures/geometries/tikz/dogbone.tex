% Illustrative construction input in mm. Labels stay symbolic.
\newcommand{\DogboneLength}{170}         % length
\newcommand{\DogboneWidth}{112}          % width
\newcommand{\DogboneCentralLength}{46}   % dogbone_central_length
\newcommand{\DogboneCentralWidth}{52}    % dogbone_central_width
\newcommand{\DogboneHeadLength}{32}      % dogbone_head_length
\newcommand{\DogboneThickness}{24}       % dogbone_thickness

\pgfmathsetmacro{\DogboneTransitionLength}{.5*(\DogboneLength-2*\DogboneHeadLength-\DogboneCentralLength)}
\pgfmathsetmacro{\DogboneTransitionWidth}{.5*(\DogboneWidth-\DogboneCentralWidth)}
\ifdim\DogboneTransitionLength pt>0pt\else
  \PackageError{civita-dogbone}{Transition length must be positive}{Check the independent dimensions.}
\fi
\ifdim\DogboneTransitionWidth pt>0pt\else
  \PackageError{civita-dogbone}{Transition width must be positive}{Check the independent dimensions.}
\fi
\ifdim\DogboneTransitionLength pt=\DogboneTransitionWidth pt\else
  \PackageError{civita-dogbone}{The prototype uses equal transition extents}{Select an equal-extents drawing fixture for the quarter-round profile.}
\fi

% O lies on z=0. The 0.43 and 0.59 visual offsets follow the supplied drawing.
% The library does not specify exact x and y offsets for that point.
\pgfmathsetmacro{\DogboneLeft}{-.43*\DogboneWidth}
\pgfmathsetmacro{\DogboneRight}{\DogboneLeft+\DogboneWidth}
\pgfmathsetmacro{\DogboneFar}{-.59*\DogboneThickness}
\pgfmathsetmacro{\DogboneNear}{\DogboneFar+\DogboneThickness}
\pgfmathsetmacro{\DogboneGaugeLeft}{\DogboneLeft+\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneGaugeRight}{\DogboneRight-\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneGaugeStart}{\DogboneHeadLength+\DogboneTransitionLength}
\pgfmathsetmacro{\DogboneGaugeEnd}{\DogboneGaugeStart+\DogboneCentralLength}
\pgfmathsetmacro{\DogboneSecondHead}{\DogboneLength-\DogboneHeadLength}
\pgfmathsetmacro{\DogboneGaugeMid}{.5*(\DogboneGaugeStart+\DogboneGaugeEnd)}

% The reference uses quarter-round arcs. The chosen fixture has equal axial
% and lateral transition extents. Use the standard cubic representation of
% each quarter arc and mirror it at the specimen midplanes.
\pgfmathsetmacro{\DogboneKappa}{4*(sqrt(2)-1)/3}
\pgfmathsetmacro{\DogboneControlLeft}{\DogboneLeft+\DogboneKappa*\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneControlRight}{\DogboneRight-\DogboneKappa*\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneControlStart}{\DogboneGaugeStart-\DogboneKappa*\DogboneTransitionLength}
\pgfmathsetmacro{\DogboneControlEnd}{\DogboneGaugeEnd+\DogboneKappa*\DogboneTransitionLength}
\newcommand{\DogboneOutline}[1]{%
  (\DogboneLeft,#1,0) -- (\DogboneRight,#1,0) -- (\DogboneRight,#1,\DogboneHeadLength)
    .. controls (\DogboneControlRight,#1,\DogboneHeadLength) and
      (\DogboneGaugeRight,#1,\DogboneControlStart) .. (\DogboneGaugeRight,#1,\DogboneGaugeStart)
  -- (\DogboneGaugeRight,#1,\DogboneGaugeEnd)
    .. controls (\DogboneGaugeRight,#1,\DogboneControlEnd) and
      (\DogboneControlRight,#1,\DogboneSecondHead) .. (\DogboneRight,#1,\DogboneSecondHead)
  -- (\DogboneRight,#1,\DogboneLength) -- (\DogboneLeft,#1,\DogboneLength)
  -- (\DogboneLeft,#1,\DogboneSecondHead)
    .. controls (\DogboneControlLeft,#1,\DogboneSecondHead) and
      (\DogboneGaugeLeft,#1,\DogboneControlEnd) .. (\DogboneGaugeLeft,#1,\DogboneGaugeEnd)
  -- (\DogboneGaugeLeft,#1,\DogboneGaugeStart)
    .. controls (\DogboneGaugeLeft,#1,\DogboneControlStart) and
      (\DogboneControlLeft,#1,\DogboneHeadLength) .. (\DogboneLeft,#1,\DogboneHeadLength)
  -- cycle%
}

% For the isometric view, two of the shoulder bands are visible up to the
% 45-degree tangent. Split their profile cubics at t=1/2 to keep the
% silhouette and projected face boundaries consistent with the outline.
\pgfmathsetmacro{\DogboneHalfNear}{.5-3*\DogboneKappa/8}
\pgfmathsetmacro{\DogboneHalfFar}{.5+3*\DogboneKappa/8}
\pgfmathsetmacro{\DogboneSplitRightX}{\DogboneGaugeRight+\DogboneHalfNear*\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneSplitRightZ}{\DogboneGaugeEnd+\DogboneHalfFar*\DogboneTransitionLength}
\pgfmathsetmacro{\DogboneSplitLeftX}{\DogboneLeft+\DogboneHalfFar*\DogboneTransitionWidth}
\pgfmathsetmacro{\DogboneSplitLeftZ}{\DogboneHeadLength+\DogboneHalfNear*\DogboneTransitionLength}

% Start, two cubic controls, and end are supplied as x-z pairs.
\newcommand{\DogboneCurveBand}[8]{%
  \path[civita/side] ({#1},\DogboneFar,{#2})
    .. controls ({#3},\DogboneFar,{#4}) and ({#5},\DogboneFar,{#6}) .. ({#7},\DogboneFar,{#8})
    -- ({#7},\DogboneNear,{#8})
    .. controls ({#5},\DogboneNear,{#6}) and ({#3},\DogboneNear,{#4}) .. ({#1},\DogboneNear,{#2})
    -- cycle;
  \draw[civita/edge] ({#1},\DogboneFar,{#2})
    .. controls ({#3},\DogboneFar,{#4}) and ({#5},\DogboneFar,{#6}) .. ({#7},\DogboneFar,{#8});
}

\newcommand{\DogboneFlatBand}[3]{%
  \path[civita/side] (#1,\DogboneFar,#2) -- (#1,\DogboneFar,#3)
    -- (#1,\DogboneNear,#3) -- (#1,\DogboneNear,#2) -- cycle;
  \draw[civita/edge] (#1,\DogboneFar,#2) -- (#1,\DogboneFar,#3);
}

\newcommand{\DogboneOrthoScale}{.46}
\newcommand{\DogboneIsoScale}{.50}

\pgfmathsetmacro{\DogboneIsoDown}{\DogboneIsoScale/sqrt(6)}
\pgfmathsetmacro{\DogboneIsoUp}{2*\DogboneIsoDown}
